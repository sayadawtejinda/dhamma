const generateContent = async (prompt, systemInstruction) => {
    const payload = {
        contents: [{ parts: [{ text: prompt }] }],
        systemInstruction: { parts: [{ text: systemInstruction }] },
        generationConfig: { responseMimeType: "application/json" }
    };

    const MAX_TRIES = 4;
    for (let attempt = 0; attempt < MAX_TRIES; attempt++) {
        let status = 0;
        try {
            const response = await fetch(API_URL, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            });
            status = response.status;
            if (!response.ok) throw new Error(`HTTP error! status: ${response.status}`);
            const result = await response.json();
            const text = result.candidates?.[0]?.content?.parts?.[0]?.text;
            if (text) return JSON.parse(text.replace(/^```json\s*|\s*```$/g, '').trim());
            throw new Error("No content");
        } catch (e) {
            if (attempt === MAX_TRIES - 1) throw e;
            // 429 / 503 = Gemini asks us to slow down: wait 10, 20, 30 seconds. Anything else: 1 second as before.
            await new Promise(r => setTimeout(r, (status === 429 || status === 503) ? 10000 * (attempt + 1) : 1000));
        }
    }
};

const VARIANT_SYSTEM = "You are an expert curriculum developer. You strictly return JSON. You ALWAYS include the exact image filenames from the source text into your generated text.";

// Asks Gemini; checks the answer against the age rules; when a real problem is found (not only wording) asks ONE more time, telling Gemini
// what was wrong. The better of the two answers is kept. Returns { data, problems } so the caller can say what is still not perfect.
const generateSingleVariantChecked = async (title, content, group) => {
    let best = null;
    let fix = '';
    for (let attempt = 0; attempt < 2; attempt++) {
        const data = await generateContent(buildVariantPrompt(title, content, group, fix), VARIANT_SYSTEM);
        const problems = checkVariant(data, group, content);
        const hard = problems.filter(p => !isSoftProblem(p));
        if (!best || hard.length < best.hard) best = { data, problems, hard: hard.length };
        if (!hard.length) break;
        fix = `\n    YOUR PREVIOUS ANSWER WAS REJECTED because: ${hard.join('; ')}. Write the whole lesson again and fix this. Respect every limit.`;
    }
    return { data: best.data, problems: best.problems };
};

const generateLessonVariants = async (title, content, mode) => {
    let targetGroups = mode === 'junior' ? ['storytellers', 'explorers'] : ['adventurers', 'voyagers'];
    const results = {};
    
    const promises = targetGroups.map(async (g) => {
        const data = await generateSingleVariant(title, content, g);
        return { group: g, data };
    });
    
    const completed = await Promise.all(promises);
    completed.forEach(item => {
        results[item.group] = item.data;
    });
    
    return results;
};

const ABHI_SPECS = {
    storytellers: { minWords: 60, maxWords: 120, maxSentenceWords: 10, maxQuestionWords: 12 },
    explorers:    { minWords: 150, maxWords: 250, maxSentenceWords: 14, maxQuestionWords: 16 },
    adventurers:  { minWords: 250, maxWords: 350, maxSentenceWords: 20, maxQuestionWords: 22 },

const wordCount = (t) => (String(t || '').trim().match(/\S+/g) || []).length;
const IMAGE_NAME_RE = /\b[\w-]+\.(?:png|jpe?g|webp|gif)\b/gi;
const imageNamesIn = (t) => Array.from(new Set(String(t || '').match(IMAGE_NAME_RE) || []));
// the English text without image names, links and markdown marks (they are not words to read)
const plainEnglish = (t) => String(t || '').replace(IMAGE_NAME_RE, ' ').replace(/https?:\/\/\S+/g, ' ').replace(/[*=_#`]+/g, '');
const sentencesOf = (t) => plainEnglish(t).split(/(?<=[.!?])\s+/).map(s => s.trim()).filter(Boolean);

// A list of problems with one generated age version (an empty list = fine).
const checkVariant = (data, group, baseContent) => {
    const spec = ABHI_SPECS[group];
    const problems = [];
    if (!data || typeof data !== 'object') return ['Gemini did not return a lesson'];
    const english = plainEnglish(data.english);
    const words = wordCount(english);
    if (words > spec.maxWords) problems.push(`the English text has ${words} words; the maximum is ${spec.maxWords}`);
    if (words < spec.minWords * 0.8) problems.push(`the English text has only ${words} words; it should have ${spec.minWords} to ${spec.maxWords}`);
    const longest = Math.max(0, ...sentencesOf(data.english).map(wordCount));
    if (longest > spec.maxSentenceWords + 4) problems.push(`one English sentence has ${longest} words; sentences must be at most ${spec.maxSentenceWords} words`);
    const needed = imageNamesIn(baseContent);
    const lostB = needed.filter(n => !String(data.burmese || '').includes(n));
    const lostE = needed.filter(n => !String(data.english || '').includes(n));
    if (lostB.length) problems.push(`the Burmese text lost these images: ${lostB.join(', ')}`);
    if (lostE.length) problems.push(`the English text lost these images: ${lostE.join(', ')}`);
    if (!data.englishTitle || !String(data.englishTitle).trim()) problems.push('there is no English title');
    if (!Array.isArray(data.discussionQuestions) || data.discussionQuestions.length !== 10) problems.push(`there are ${Array.isArray(data.discussionQuestions) ? data.discussionQuestions.length : 0} discussion questions; there must be exactly 10`);
    const qs = data.quiz && Array.isArray(data.quiz.questions) ? data.quiz.questions : [];
    if (qs.length !== 10) problems.push(`the quiz has ${qs.length} questions; there must be exactly 10`);
    qs.forEach((q, i) => {
        const opts = Array.isArray(q.options) ? q.options.map(o => String(o).trim().toLowerCase()) : [];
        if (opts.length !== 4 || new Set(opts).size !== 4) problems.push(`quiz question ${i + 1} must have 4 different options`);
        else if (!Number.isInteger(q.correctAnswerIndex) || q.correctAnswerIndex < 0 || q.correctAnswerIndex > 3) problems.push(`quiz question ${i + 1} has no correct answer`);
        if (wordCount(q.question) > spec.maxQuestionWords + 6) problems.push(`quiz question ${i + 1} is too long for this age`);
    });
    return problems;
};

const isSoftProblem = (p) => /one English sentence has|too long for this age/.test(p);

// ---- reading the lessons file (many lessons at once) ----
// a pasted chat answer has no # or === lines: a line that starts with "Lesson 3", "Article 3", "ဆောင်းပါး (၃)" or "သင်ခန်းစာ ၃" starts a lesson
const BULK_HEAD_RE = /^[*#\s]*(?:ဆောင်းပါး|သင်ခန်းစာ|article|lesson|chapter)\s*\(?\s*[0-9၀-၉]+\s*\)?\s*[-–—:.။)]?/i;
const bulkCleanTitle = (x) => String(x || '').replace(/\*\*|__/g, '').replace(/^(ဆောင်းပါး|article|သင်ခန်းစာ|lesson)\s*\(?\s*[0-9၀-၉]+\s*\)?\s*[-–—:.။]?\s*/i, '').trim();
// Markdown -> plain text: no #, no **, no ---, tables as lines of " – " separated cells.
const bulkMdToPlain = (md) => {
    const out = [];
    for (const raw of String(md || '').replace(/\r/g, '').split('\n')) {
        let line = raw.trimEnd();
        const t = line.trim();
        if (/^([-*_])(\s*\1){2,}$/.test(t)) { out.push(''); continue; }
        if (/^\|?[\s:|-]+\|?$/.test(t) && t.includes('-') && t.includes('|')) continue;
        if (/^\|.*\|$/.test(t)) line = t.replace(/^\||\|$/g, '').split('|').map(c => c.trim()).filter(Boolean).join(' – ');
        line = line.replace(/^\s{0,3}#{1,6}\s+/, '').replace(/^\s*>\s?/, '');
        line = line.replace(/\*\*(.+?)\*\*/g, '$1').replace(/__(.+?)__/g, '$1');
        out.push(line);
    }
    return out.join('\n').replace(/\n{3,}/g, '\n\n').trim();
};
// One file -> [{ title, content }].
//   .json            : { lessons: [ { title, burmeseContent } ] }  (what the make-lessons tool writes), or the app's own exported lessons
//   text / markdown  : a lesson starts with a line  "# Title"  or  "=== Title ===" ; no such line = the whole file is one lesson
const parseBulkFile = (fileName, text) => {
    if (/\.json$/i.test(fileName)) {
        const data = JSON.parse(String(text || '').replace(/^﻿/, ''));
        const list = Array.isArray(data) ? data : (data.lessons || []);
        return list.map(l => ({ title: String(l.title || ''), content: String(l.burmeseContent || l.content || '') })).filter(d => d.content.length > 40);
    }
    const lines = String(text || '').replace(/^﻿/, '').replace(/\r/g, '').split('\n');
    const marker = lines.some(l => /^\s*===\s*\S/.test(l));
    const h1 = !marker && lines.some(l => /^#\s+\S/.test(l));
    const numbered = !marker && !h1 && lines.some(l => BULK_HEAD_RE.test(l.trim()));
    const stories = []; let cur = null;
    for (const raw of lines) {
        const line = raw.trim();
        if (marker && /^===\s*\S/.test(line)) { cur = { heading: bulkCleanTitle(line.replace(/^=+\s*/, '').replace(/\s*=+$/, '')), body: [] }; stories.push(cur); continue; }
        if (h1 && /^#\s+\S/.test(raw)) { cur = { heading: bulkCleanTitle(raw.replace(/^#\s+/, '')), body: [] }; stories.push(cur); continue; }
        if (numbered && BULK_HEAD_RE.test(line)) { cur = { heading: bulkCleanTitle(line.replace(/^[*#\s]+|[*\s]+$/g, '')), body: [] }; stories.push(cur); continue; }
        if (cur) cur.body.push(raw);
        else if (!marker && !h1 && !numbered) { if (!stories.length) { cur = { heading: '', body: [] }; stories.push(cur); } cur.body.push(raw); }
    }
    return stories.map(s => ({ title: s.heading, content: bulkMdToPlain(s.body.join('\n')) })).filter(d => d.content.length > 40);
};

const AudioPlayer = ({ src }) => {
    const [playing, setPlaying] = useState(false);
    const audioRef = useRef(null);

    const togglePlay = () => {
        if (!audioRef.current) {
            audioRef.current = new Audio(src);
            audioRef.current.onended = () => setPlaying(false);
        }
        if (playing) audioRef.current.pause();
        else audioRef.current.play().catch(e => console.error(e));
        setPlaying(!playing);
    };

    return (
        <button onClick={togglePlay} className="inline-flex items-center gap-1 px-2 py-1 bg-pink-600/20 text-pink-400 rounded-full hover:bg-pink-600/40 transition border border-pink-500/30 mx-1">
            {playing ? <Pause className="w-4 h-4" /> : <Music className="w-4 h-4" />}
            <span className="text-xs font-bold">Play</span>
        </button>
    );
};

const SmartContent = ({ text }) => {
    if (!text) return null;
    const formatText = (content) => {
        const parts = content.split(/(\*\*.*?\*\*|\*.*?\*|==.*?==)/g);
        return parts.map((part, idx) => {
            if (part.startsWith('**') && part.endsWith('**')) return <strong key={idx} className="text-yellow-200 font-bold">{part.slice(2, -2)}</strong>;
            if (part.startsWith('*') && part.endsWith('*')) return <em key={idx} className="text-indigo-300 italic">{part.slice(1, -1)}</em>;
            if (part.startsWith('==') && part.endsWith('==')) return <span key={idx} className="bg-yellow-600/40 px-1 rounded text-white border border-yellow-500/30 mx-0.5 shadow-sm box-decoration-clone">{part.slice(2, -2)}</span>;
            return part;
        });
    };

    const regex = /((?:https?:\/\/[^\s]+)|(?:\b[\w-]+\.(?:png|jpg|jpeg|gif|mp3)\b))/gi;
    const parts = text.split(regex);

    return (
        <span className="leading-relaxed">
            {parts.map((part, i) => {
                if (!part) return null;
                const isUrl = part.match(/^https?:\/\//i);
                const isImage = part.match(/\.(png|jpg|jpeg|gif)$/i);
                const isAudio = part.match(/\.mp3$/i);

                if (isUrl) {
                    if (part.match(/(youtube\.com|youtu\.be)/i)) return <a key={i} href={part} target="_blank" rel="noopener noreferrer" className="inline-flex items-center text-red-400 hover:text-red-300 mx-1 font-semibold"><Youtube className="w-5 h-5 mr-1" /> Video</a>;
                    if (isAudio) return <AudioPlayer key={i} src={part} />;
                    if (isImage) return <div key={i} className="my-2"><img src={part} className="max-w-full h-auto rounded-lg shadow-md border border-gray-600 mx-auto" onError={(e) => e.target.style.display = 'none'} /></div>;
                    return <a key={i} href={part} target="_blank" rel="noopener noreferrer" className="text-blue-400 hover:underline inline-flex items-center mx-1">Link <ExternalLink className="w-3 h-3 ml-1" /></a>;
                }
                
               if (isImage) return <div key={i} className="my-2"><img src={`${IMG_BASE_URL}${part}`} className="max-w-full h-auto rounded-lg shadow-md border border-gray-600 mx-auto" onError={(e) => e.target.style.display = 'none'} /></div>;
                if (isAudio) return <AudioPlayer key={i} src={`${AUDIO_BASE_URL}${part}`} />;
                
                return <span key={i}>{formatText(part)}</span>;
            })}
        </span>
    );
};

const NotificationBell = ({ userId }) => {
    const [notifications, setNotifications] = useState([]);
    const [isOpen, setIsOpen] = useState(false);
    const [lastRead, setLastRead] = useState(() => {
        return parseInt(localStorage.getItem(`last_read_notif_${userId}`)) || 0;
    });

    useEffect(() => {
        if (!db || !userId) return;
        const q = query(collection(db, 'artifacts', appId, 'public', 'data', 'activity_feed'), orderBy('timestamp', 'desc'), limit(15));
        const unsub = onSnapshot(q, (snap) => {
            const notifs = snap.docs.map(d => ({ id: d.id, ...d.data() }));
            setNotifications(notifs);
        });
        return () => unsub();
    }, [userId]);

    const unreadCount = notifications.filter(n => {
        const ts = n.timestamp?.toMillis ? n.timestamp.toMillis() : (n.timestamp?.seconds * 1000) || 0;
        return ts > lastRead;
    }).length;

    const handleToggle = () => {
        if (!isOpen) {
            const now = Date.now();
            setLastRead(now);
            localStorage.setItem(`last_read_notif_${userId}`, now);
        }
        setIsOpen(!isOpen);
    };

    return (
        <div className="relative">
            <button onClick={handleToggle} className="p-2 bg-gray-700 hover:bg-gray-600 rounded-full transition relative shadow-lg">
                <Bell className="w-5 h-5 text-gray-300" />
                {unreadCount > 0 && (
                    <span className="absolute -top-1 -right-1 bg-red-500 text-white text-[10px] font-bold px-1.5 py-0.5 rounded-full animate-pulse shadow-md">
                        {unreadCount > 9 ? '9+' : unreadCount}
                    </span>
                )}
            </button>

            {isOpen && (
                <div className="absolute right-0 mt-2 w-72 max-w-[90vw] bg-gray-800 border border-gray-600 rounded-xl shadow-2xl z-50 overflow-hidden animate-bounce-in">
                    <div className="p-3 bg-gray-750 border-b border-gray-700 flex justify-between items-center">
                        <h4 className="font-bold text-white text-sm flex items-center gap-2">
                            <Bell className="w-4 h-4 text-indigo-400" /> Notifications
                        </h4>
                        <button onClick={() => setIsOpen(false)} className="text-gray-400 hover:text-white"><X className="w-4 h-4"/></button>
                    </div>
                    <div className="max-h-64 overflow-y-auto custom-scrollbar p-2 space-y-2">
                        {notifications.length === 0 ? (
                            <p className="text-center text-gray-500 text-xs py-6 italic">No new notifications</p>
                        ) : (
                            notifications.map(n => (
                                <div key={n.id} className="bg-gray-700/50 p-3 rounded-lg border border-gray-600/50 flex items-start gap-3">
                                    <div className="mt-1"><Trophy className="w-4 h-4 text-yellow-400" /></div>
                                    <div>
                                        <p className="text-xs font-bold text-white"><span className="text-indigo-300">{n.studentName}</span> finished a quiz!</p>
                                        <p className="text-[10px] text-gray-400 truncate mt-0.5">{n.lessonTitle}</p>
                                    </div>
                                </div>
                            ))
                        )}
                    </div>
                </div>
            )}
        </div>
    );
};

const QuizModule = ({ lessonId, lessonTitle, userId, userName, ageGroup, quizData, onClose }) => {
    const [gameState, setGameState] = useState('playing'); 
    const [currentQIndex, setCurrentQIndex] = useState(0);
    const [score, setScore] = useState(0);
    const [timeLeft, setTimeLeft] = useState(30);
    const [showOptions, setShowOptions] = useState(false);
    
    const [selectedOption, setSelectedOption] = useState(null);
    const [feedbackStatus, setFeedbackStatus] = useState(null);
    const [isProcessing, setIsProcessing] = useState(false);
const [correctCount, setCorrectCount] = useState(0);

    useEffect(() => {
        if (gameState !== 'playing' || showOptions) return;
        const t1 = setTimeout(() => setShowOptions(true), 5000); 
        return () => clearTimeout(t1);
    }, [gameState, showOptions, currentQIndex]);

    useEffect(() => {
        if (gameState !== 'playing' || !showOptions || isProcessing) return;

        const timer = setInterval(() => {
            setTimeLeft(prev => prev - 1);
        }, 1000);
        return () => clearInterval(timer);
    }, [gameState, showOptions, isProcessing, currentQIndex]);

    useEffect(() => {
        if (timeLeft <= 0 && showOptions && !isProcessing && gameState === 'playing') {
            handleAnswer(-1, true);
        }
    }, [timeLeft, showOptions, isProcessing, gameState]);

    const handleAnswer = async (optionIndex, isTimeout = false) => {
        if (isProcessing) return;
        setIsProcessing(true);
        setSelectedOption(optionIndex);

        const currentQ = quizData.questions[currentQIndex];
        let isCorrect = false;
        if (!isTimeout && currentQ.correctAnswerIndex !== undefined) isCorrect = optionIndex === currentQ.correctAnswerIndex;

        setFeedbackStatus(isCorrect ? 'correct' : 'incorrect');

        let points = 0;
        if (isCorrect) points = Math.max(100, Math.ceil((timeLeft / 30) * 1000));
        const newCorrectCount = isCorrect ? correctCount + 1 : correctCount;
        if (isCorrect) setCorrectCount(newCorrectCount);
        
        setTimeout(async () => {
            setScore(prev => prev + points);
            setFeedbackStatus(null); 
            setSelectedOption(null); 
            setIsProcessing(false);

            if (currentQIndex + 1 < quizData.questions.length) {
                setCurrentQIndex(prev => prev + 1); 
                setShowOptions(false); 
                setTimeLeft(30);
            } else {
                const finalScore = score + points;
                const PASS_MARK = 8;

                if (newCorrectCount < PASS_MARK) {
                    setGameState('failed');
                    return;
                }

                setGameState('finished');
                window._quizCorrectCount = 0;
                
                await addDoc(collection(db, 'artifacts', appId, 'public', 'data', 'lessons', lessonId, 'quiz', ageGroup, 'results'), {
                    name: userName, score: finalScore, userId: userId, group: ageGroup, timestamp: serverTimestamp()
                });

                const globalDocId = `${userId}_${lessonId}_${ageGroup}`;
                const globalDocRef = doc(db, 'artifacts', appId, 'public', 'data', 'global_scores', globalDocId);
                try {
                    const docSnap = await getDoc(globalDocRef);
                    if (!docSnap.exists()) {
                        await setDoc(globalDocRef, { userId, name: userName, group: ageGroup, score: finalScore, lessonId, timestamp: serverTimestamp() });
                    }
                } catch (e) { console.error("Error saving global score", e); }

                try {
                    await addDoc(collection(db, 'artifacts', appId, 'public', 'data', 'activity_feed'), {
                        type: 'quiz_completed',
                        studentName: userName,
                        lessonTitle: lessonTitle,
                        group: ageGroup,
                        timestamp: serverTimestamp()
                    });
                } catch (e) { console.error("Error posting notification", e); }
            }
        }, 1500);
    };
if (gameState === 'failed') {
        return (
            <div className="fixed inset-0 bg-gray-900 z-50 flex items-center justify-center p-4">
                <div className="bg-gray-800 p-8 rounded-2xl text-center shadow-2xl animate-bounce-in border border-red-500">
                    <div className="text-6xl mb-4">😔</div>
                    <h2 className="text-3xl font-bold text-red-400 mb-2">Not Enough Correct!</h2>
                    <p className="text-gray-300 text-lg mb-2">You need at least <span className="text-yellow-400 font-black">8 correct answers</span> to pass.</p>
                    <p className="text-gray-400 mb-6">Please read the lesson again and try your best!</p>
                    <button onClick={onClose} className="bg-indigo-600 hover:bg-indigo-700 text-white font-bold py-3 px-8 rounded-full shadow-lg transition transform hover:scale-105">
                        Go Back & Review
                    </button>
                </div>
            </div>
        );
    }
    if (gameState === 'finished') {
        return (
             <div className="fixed inset-0 bg-gray-900 z-50 flex items-center justify-center p-4">
                <div className="bg-gray-800 p-8 rounded-2xl text-center shadow-2xl animate-bounce-in">
                    <Trophy className="w-20 h-20 text-yellow-400 mx-auto mb-4" />
                    <h2 className="text-3xl font-bold text-white mb-2">Quiz Completed!</h2>
                    <p className="text-gray-400 text-lg mb-6">Your Score: <span className="text-indigo-400 font-bold">{score}</span></p>
                    <button onClick={onClose} className="bg-green-600 hover:bg-green-700 text-white font-bold py-3 px-8 rounded-full shadow-lg transition transform hover:scale-105">
                        Finish & Mark Complete
                    </button>
                </div>
             </div>
        );
    }

    return (
        <div className="fixed inset-0 bg-gray-900 z-50 flex flex-col items-center justify-center p-4 overflow-y-auto">
            <button onClick={onClose} className="absolute top-4 right-4 text-gray-400 hover:text-white"><X className="w-8 h-8"/></button>
            <div className="w-full max-w-3xl flex flex-col h-full justify-center">
                <div className="flex justify-between items-center mb-6 px-4">
                    <div className="bg-gray-800 px-4 py-2 rounded-full border border-gray-600 text-white font-bold">Q {currentQIndex + 1}/{quizData.questions.length}</div>
                    {showOptions && <div className="bg-yellow-500 text-black px-6 py-2 rounded-full font-black text-xl shadow-lg flex items-center"><Timer className="w-5 h-5 mr-2" /> {timeLeft}s</div>}
                    <div className="bg-indigo-600 px-4 py-2 rounded-full text-white font-bold">Score: {score}</div>
                </div>

                <div className="bg-white text-black p-8 rounded-xl shadow-2xl mb-8 text-center min-h-[200px] flex items-center justify-center relative overflow-hidden">
                    <h2 className="text-2xl md:text-3xl font-bold z-10">{quizData.questions[currentQIndex].question}</h2>
                    {feedbackStatus && (
                        <div className={`absolute inset-0 flex items-center justify-center z-20 ${feedbackStatus === 'correct' ? 'bg-green-100/90 text-green-700' : 'bg-red-100/90 text-red-700'}`}>
                            <div className="text-4xl font-black animate-bounce-in">{feedbackStatus === 'correct' ? "CORRECT!" : "WRONG!"}</div>
                        </div>
                    )}
                </div>

                {showOptions ? (
                    <div className={`grid gap-4 ${quizData.questions[currentQIndex].options.length === 2 ? 'grid-cols-2' : 'grid-cols-1 md:grid-cols-2'}`}>
                        {quizData.questions[currentQIndex].options.map((opt, idx) => {
                            let btnClass = "hover:scale-105 active:scale-95";
                            let bgClass = ['bg-red-500', 'bg-blue-500', 'bg-yellow-500', 'bg-green-500'][idx % 4];
                            if (feedbackStatus) {
                                btnClass = "cursor-not-allowed opacity-50"; 
                                if (idx === quizData.questions[currentQIndex].correctAnswerIndex) bgClass = "bg-green-600 ring-4 ring-green-300 opacity-100"; 
                                else if (idx === selectedOption && feedbackStatus === 'incorrect') bgClass = "bg-red-600 ring-4 ring-red-300 opacity-100"; 
                            }
                            return (
                                <button key={idx} onClick={() => handleAnswer(idx)} disabled={isProcessing} className={`${bgClass} text-white font-bold text-lg p-6 rounded-xl shadow-lg transform transition flex items-center ${btnClass}`}>
                                    <span className="bg-black/20 w-8 h-8 rounded flex items-center justify-center mr-4">{['A','B','C','D'][idx]}</span>{opt}
                                </button>
                            );
                        })}
                    </div>
                ) : <div className="text-center py-20"><div className="animate-pulse text-4xl font-black text-yellow-400">Get Ready... (5s)</div></div>}
            </div>
        </div>
    );
};

const QAList = ({ lessonId, isTeacher, userId, userName, suggestedQuestions = [], lessonContext = "" }) => {
    const [questions, setQuestions] = useState([]);
    const [newQuestion, setNewQuestion] = useState('');
    const [replyInputs, setReplyInputs] = useState({});
    const [isSubmitting, setIsSubmitting] = useState(false);
    const [aiGenerating, setAiGenerating] = useState(null);
    const [visibleSuggestions, setVisibleSuggestions] = useState([]);
    const [suggestionPool, setSuggestionPool] = useState([]);
    const topRef = useRef(null);
    const bottomRef = useRef(null);

    useEffect(() => {
        if (suggestedQuestions.length > 0) {
            setVisibleSuggestions(suggestedQuestions.slice(0, 3));
            setSuggestionPool(suggestedQuestions.slice(3));
        }
    }, [suggestedQuestions]);

    useEffect(() => {
        if (!db) return;
        const qRef = collection(db, 'artifacts', appId, 'public', 'data', 'lessons', lessonId, 'questions');
        const q = query(qRef, orderBy('timestamp', 'asc'));
        return onSnapshot(q, (snap) => setQuestions(snap.docs.map(d => ({ id: d.id, ...d.data() }))));
    }, [lessonId]);

    const handleUseSuggestion = (suggestion, index) => {
        setNewQuestion(suggestion);
        if (suggestionPool.length > 0) {
            const randomIndex = Math.floor(Math.random() * suggestionPool.length);
            const newVisible = [...visibleSuggestions];
            newVisible[index] = suggestionPool[randomIndex];
            const newPool = [...suggestionPool];
            newPool.splice(randomIndex, 1);
            setVisibleSuggestions(newVisible); setSuggestionPool(newPool);
        }
        bottomRef.current?.scrollIntoView({ behavior: 'smooth' });
    };

    const handleAsk = async (textToAsk = newQuestion) => {
        if (!textToAsk.trim() || isSubmitting) return;
        setIsSubmitting(true);
        try {
            let translation = '';
            let originalLang = isTeacher ? 'mm' : 'en';
            try { const res = await translateText(textToAsk, isTeacher ? 'en' : 'mm'); translation = res.translatedText; } catch (err) { translation = textToAsk; }
            await addDoc(collection(db, 'artifacts', appId, 'public', 'data', 'lessons', lessonId, 'questions'), {
                studentId: userId, studentName: userName || "Anonymous", text: textToAsk, translation, originalLang, timestamp: serverTimestamp(), replies: [], likes: []
            });
            setNewQuestion('');
        } catch (e) { console.error(e); } finally { setIsSubmitting(false); }
    };

    const handleReply = async (questionId) => {
        const text = replyInputs[questionId];
        if (!text?.trim() || isSubmitting) return;
        setIsSubmitting(true);
        try {
            let translation = '';
            let originalLang = isTeacher ? 'mm' : 'en';
            try { const res = await translateText(text, isTeacher ? 'en' : 'mm'); translation = res.translatedText; } catch(err) { translation = text; }
            const reply = { id: crypto.randomUUID(), userId, userName: userName || (isTeacher ? "Teacher" : "Student"), role: isTeacher ? 'Teacher' : 'Student', text: text.trim(), translation, originalLang, timestamp: new Date().toISOString(), likes: [] };
            await updateDoc(doc(db, 'artifacts', appId, 'public', 'data', 'lessons', lessonId, 'questions', questionId), { replies: arrayUnion(reply) });
            setReplyInputs(prev => ({ ...prev, [questionId]: '' }));
        } catch (e) { console.error(e); } finally { setIsSubmitting(false); }
    };

    const generateAIReply = async (qId, questionText) => {
        setAiGenerating(qId);
        try {
            const contextSnippet = lessonContext ? lessonContext.substring(0, 1500) : "No context available";
            const prompt = `Context: "${contextSnippet}"\nStudent Question: "${questionText}"\nInstruction: Provide a brief, helpful, educational answer. Return JSON: { "reply": "Your string" }`;
            const jsonRes = await generateContent(prompt, "You are a helpful teacher assistant.");
            setReplyInputs(prev => ({ ...prev, [qId]: jsonRes.reply }));
        } catch(e) { console.error(e); } finally { setAiGenerating(null); }
    };

    const toggleLike = async (qId, reply = null) => {
        const docRef = doc(db, 'artifacts', appId, 'public', 'data', 'lessons', lessonId, 'questions', qId);
        try {
            const snap = await getDoc(docRef);
            const data = snap.data();
            if (reply) {
                const updatedReplies = data.replies.map(r => {
                    if (r.id === reply.id) {
                        const likes = r.likes || [];
                        return { ...r, likes: likes.includes(userId) ? likes.filter(id => id !== userId) : [...likes, userId] };
                    }
                    return r;
                });
                await updateDoc(docRef, { replies: updatedReplies });
            } else {
                const likes = data.likes || [];
                await updateDoc(docRef, { likes: likes.includes(userId) ? likes.filter(id => id !== userId) : [...likes, userId] });
            }
        } catch(e) { console.error(e); }
    };

    const renderText = (item) => {
        let displayContent = item.text;
        if (isTeacher && item.originalLang === 'en') displayContent = item.translation || item.text;
        else if (!isTeacher && item.originalLang === 'mm') displayContent = item.translation || item.text;
        return <SmartContent text={displayContent} />;
    };

    return (
        <div className="space-y-6 mt-4 relative">
            <div ref={topRef}></div>
            <div className="fixed bottom-10 right-6 flex flex-col gap-3 z-40">
                <button onClick={() => topRef.current?.scrollIntoView({ behavior: 'smooth' })} className="p-3 bg-indigo-600/90 hover:bg-indigo-500 text-white rounded-full shadow-2xl backdrop-blur transition-transform hover:scale-110 border border-indigo-400/50" title="Scroll to Top"><ArrowUp className="w-5 h-5"/></button>
                <button onClick={() => bottomRef.current?.scrollIntoView({ behavior: 'smooth' })} className="p-3 bg-indigo-600/90 hover:bg-indigo-500 text-white rounded-full shadow-2xl backdrop-blur transition-transform hover:scale-110 border border-indigo-400/50" title="Scroll to Bottom"><ArrowDown className="w-5 h-5"/></button>
            </div>

            {!isTeacher && visibleSuggestions.length > 0 && (
                <div className="bg-indigo-900/30 p-4 rounded-xl border border-indigo-500/30 mb-6 transition-all duration-300">
                    <h4 className="text-indigo-300 text-sm font-bold mb-3 flex items-center gap-2"><Sparkles className="w-4 h-4"/> Curious? Try asking:</h4>
                    <div className="flex flex-wrap gap-2">
                        {visibleSuggestions.map((sq, idx) => (
                            <button key={idx} onClick={() => handleUseSuggestion(sq, idx)} className="text-left text-xs bg-indigo-800/50 hover:bg-indigo-700 text-indigo-100 px-3 py-2 rounded-lg border border-indigo-600 transition hover:scale-105 active:scale-95">{sq}</button>
                        ))}
                    </div>
                </div>
            )}

            {questions.map(q => (
                <div key={q.id} className="p-4 rounded-xl bg-gray-800/50 border border-gray-700/50 shadow-sm">
                    <div className="flex justify-between items-start mb-2">
                        <div className="flex-1">
                            <div className="flex items-center gap-2 mb-1">
                                <span className="text-xs font-bold px-2 py-0.5 rounded bg-indigo-500/20 text-indigo-300 flex items-center gap-1"><User className="w-3 h-3"/> {q.studentName || "Student"}</span>
                            </div>
                            <div className="text-white font-medium">{renderText(q)}</div>
                        </div>
                        <button onClick={() => toggleLike(q.id)} className={`flex items-center gap-1 text-xs px-2 py-1 rounded transition ${q.likes?.includes(userId) ? 'text-pink-500 bg-pink-500/10' : 'text-gray-500 hover:bg-gray-700'}`}>
                            <Heart className={`w-3 h-3 ${q.likes?.includes(userId) ? 'fill-current' : ''}`} /> {q.likes?.length || 0}
                        </button>
                    </div>

                    <div className="ml-2 pl-3 border-l-2 border-gray-700 space-y-3 mt-3">
                        {q.replies && q.replies.map((reply, idx) => (
                            <div key={idx} className={`text-sm ${reply.role === 'Teacher' ? 'bg-teal-900/20 border-teal-800/50' : 'bg-gray-700/30 border-gray-600/50'} p-3 rounded-lg border`}>
                                <div className="flex justify-between items-start">
                                    <div className="flex-1">
                                        <div className="flex items-center gap-2 mb-1">
                                            <span className={`text-xs font-bold ${reply.role === 'Teacher' ? 'text-teal-400' : 'text-gray-400'} flex items-center gap-1`}>
                                                {reply.role === 'Teacher' ? <Wand2 className="w-3 h-3"/> : <User className="w-3 h-3"/>} {reply.userName || reply.role}
                                            </span>
                                        </div>
                                        <div className="text-gray-200">{renderText(reply)}</div>
                                    </div>
                                    <button onClick={() => toggleLike(q.id, reply)} className={`flex items-center gap-1 text-xs px-2 py-1 rounded transition ${reply.likes?.includes(userId) ? 'text-pink-500 bg-pink-500/10' : 'text-gray-500 hover:bg-gray-700'}`}>
                                        <Heart className={`w-3 h-3 ${reply.likes?.includes(userId) ? 'fill-current' : ''}`} /> {reply.likes?.length || 0}
                                    </button>
                                </div>
                            </div>
                        ))}
                    </div>

                    <div className="mt-3 flex flex-col gap-2">
                        {isTeacher && (!q.replies || q.replies.length === 0) && (
                            <div className="flex justify-end">
                                <button onClick={() => generateAIReply(q.id, q.translation || q.text)} className="text-xs flex items-center gap-1 text-purple-400 hover:text-purple-300 transition" disabled={aiGenerating === q.id}>
                                    {aiGenerating === q.id ? <RotateCw className="w-3 h-3 animate-spin"/> : <Wand2 className="w-3 h-3" />}
                                    {aiGenerating === q.id ? "Generating..." : "AI Suggestion"}
                                </button>
                            </div>
                        )}
                        <div className="flex gap-2">
                            <input value={replyInputs[q.id] || ''} onChange={e => setReplyInputs(prev => ({ ...prev, [q.id]: e.target.value }))} placeholder="Write a reply..." className="flex-1 bg-gray-900 text-white text-sm px-3 py-2 rounded-lg border border-gray-600 focus:border-indigo-500 focus:outline-none transition" onKeyDown={e => e.key === 'Enter' && handleReply(q.id)} />
                            <button onClick={() => handleReply(q.id)} disabled={isSubmitting} className="p-2 bg-indigo-600 rounded-lg text-white hover:bg-indigo-700 transition shadow-lg"><Send className="w-4 h-4" /></button>
                        </div>
                    </div>
                </div>
            ))}

            <div className="pt-4 border-t border-gray-700 pb-10" ref={bottomRef}>
                <p className="text-xs text-gray-400 mb-2 font-bold uppercase tracking-wider">New Question</p>
                <div className="flex gap-2">

const LessonItem = ({ lesson, isTeacher, studentAgeGroup, studentName, onGenerateVariants, onTakeQuiz, onEdit, isGenerating, userId, isOpen, onToggle }) => {

    const [showLeaderboard, setShowLeaderboard] = useState(false);
    const [leaderboardData, setLeaderboardData] = useState([]);
    const [loadingLB, setLoadingLB] = useState(false);
    const [hasUnread, setHasUnread] = useState(false);
    const [hasAsked, setHasAsked] = useState(false);
    const [hasReplied, setHasReplied] = useState(false);
    const [isCompleted, setIsCompleted] = useState(false);
    const lessonRef = useRef(null);

    useEffect(() => {
        if (isOpen && lessonRef.current) {
            setTimeout(() => {
                const y = lessonRef.current.getBoundingClientRect().top + window.scrollY - 80;
                window.scrollTo({ top: y, behavior: 'smooth' });
            }, 100);
        }
    }, [isOpen]);

    useEffect(() => {
        if (!db || !userId) return;
        const qRef = collection(db, 'artifacts', appId, 'public', 'data', 'lessons', lesson.id, 'questions');
        const unsub = onSnapshot(qRef, (snap) => {
            let unread = false; let asked = false; let replied = false;
            snap.docs.forEach(d => {
                const q = d.data();
                if (q.studentId === userId) asked = true;
                if (q.studentId !== userId && (!q.likes || !q.likes.includes(userId))) unread = true;
                if (q.replies) {
                    q.replies.forEach(r => {
                        if (r.userId === userId) replied = true;
                        if (r.userId !== userId && (!r.likes || !r.likes.includes(userId))) unread = true;
                    });
                }
            });
            setHasUnread(unread); setHasAsked(asked); setHasReplied(replied);
        });

        if (studentAgeGroup) {
            const resultRef = collection(db, 'artifacts', appId, 'public', 'data', 'lessons', lesson.id, 'quiz', studentAgeGroup, 'results');
            const unsubResult = onSnapshot(query(resultRef, where('userId', '==', userId)), (snap) => setIsCompleted(!snap.empty));
            return () => { unsub(); unsubResult(); };
        }
        return () => unsub();
    }, [lesson.id, userId, studentAgeGroup]);

            displayTitle = myVariant.englishTitle || lesson.translations?.englishTitle || lesson.title;

            discussionQuestions = myVariant.discussionQuestions || [];
        } else {

            {/* Generating Overlay */}

                    <RotateCw className="w-12 h-12 text-teal-400 animate-spin mb-4" />

                    <p className="text-gray-300 text-sm">Please wait. This usually takes about 15-30 seconds.</p>
                </div>
            )}

            <div onClick={onToggle} className="p-4 cursor-pointer hover:bg-gray-600/50 transition flex justify-between items-center">
                <div className="flex items-center gap-3">
                    {isOpen ? <ChevronDown className="w-5 h-5 text-gray-400" /> : <ChevronRight className="w-5 h-5 text-gray-400" />}
                    <div className="flex flex-col">
                        <h3 className={`text-lg font-bold flex items-center gap-2 ${isTeacher ? 'text-teal-300' : 'text-white'}`}>
                            {displayTitle}
                            {isCompleted && <span className="bg-green-500/20 text-green-400 text-xs px-2 py-0.5 rounded-full flex items-center gap-1 border border-green-500/30"><CheckCheck className="w-3 h-3"/> Completed</span>}
                            {hasUnread && <span className="w-3 h-3 bg-red-500 rounded-full animate-pulse shadow-[0_0_8px_red]"></span>}

                        {isTeacher && <span className="text-xs text-gray-500 flex gap-2">
                             {hasJunior && hasSenior ? ( <span className="text-green-400 flex items-center"><CheckCircle className="w-3 h-3 mr-1"/> Ready</span> ) : (
                                <span className="text-yellow-500 flex items-center gap-1">Pending: {!hasJunior && <span className="bg-yellow-900/40 px-1 rounded">Junior</span>} {!hasSenior && <span className="bg-yellow-900/40 px-1 rounded">Senior</span>}</span>
                             )}
                        </span>}

                </div>

const WelcomeModal = ({ userId, onStudentComplete, onTeacherComplete }) => {
    const [step, setStep] = useState('role_select');
    const [name, setName] = useState('');
    const [selectedGroup, setSelectedGroup] = useState(null);
    const [passcode, setPasscode] = useState('');
    const [error, setError] = useState('');
    const [previousName, setPreviousName] = useState(null);
    const [isSubmittingProfile, setIsSubmittingProfile] = useState(false);

    useEffect(() => {
        if (!userId || step !== 'waiting_approval') return;
        const unsub = onSnapshot(doc(db, 'artifacts', appId, 'public', 'data', 'students', userId), (snap) => {
            if (snap.exists() && snap.data().status !== 'deleted' && snap.data().status !== 'rejected') {
                const data = snap.data();
                if (data.status === 'approved') {
                    onStudentComplete({ ...data, name: data.name, group: data.group || selectedGroup });
                } else if (data.status === 'rejected') {
                    setStep('rejected');
                }
            } else {
                setStep('role_select'); 
            }
        });
        return () => unsub();
    }, [userId, step, onStudentComplete, selectedGroup]);

    const handleStudentSubmit = async () => {
        if (!name.trim() || !selectedGroup) return;
        setIsSubmittingProfile(true);
        try {
            const prefSnap = await getDoc(doc(db, 'artifacts', appId, 'public', 'data', 'settings', 'preferences'));
            const autoApprove = prefSnap.exists() ? prefSnap.data().autoApprove : false;

            const studentRef = doc(db, 'artifacts', appId, 'public', 'data', 'students', userId);
            const snap = await getDoc(studentRef);

            let initialStatus = autoApprove ? 'approved' : 'pending';
            let studentNumber = null;

            if (initialStatus === 'approved') {
                const sSnap = await getDocs(collection(db, 'artifacts', appId, 'public', 'data', 'students'));
                const approved = sSnap.docs.map(d=>d.data()).filter(s=>s.status==='approved');
                studentNumber = approved.reduce((max, s) => (s.studentNumber || 0) > max ? s.studentNumber : max, 0) + 1;
            }

            if (snap.exists() && snap.data().status !== 'deleted' && snap.data().status !== 'rejected') {
                const data = snap.data();
                if (data.status === 'approved') {
                    if (data.name.toLowerCase() === name.trim().toLowerCase()) {
                        onStudentComplete({ ...data, name: data.name, group: selectedGroup });
                        return;
                    } else {
                        setPreviousName(data.name);
                        await updateDoc(studentRef, { pendingName: name.trim(), status: initialStatus, group: selectedGroup, ...(studentNumber && {studentNumber}) });
                        setStep('waiting_approval');
                    }
                } else if (data.status === 'rejected') {
                     await updateDoc(studentRef, { name: name.trim(), status: initialStatus, group: selectedGroup, pendingName: null, ...(studentNumber && {studentNumber}) });
                     setStep('waiting_approval');
                } else {
                     await updateDoc(studentRef, { name: name.trim(), group: selectedGroup, status: initialStatus, ...(studentNumber && {studentNumber}) });
                     setStep('waiting_approval');
                }
            } else {
                await setDoc(studentRef, { userId, name: name.trim(), group: selectedGroup, status: initialStatus, timestamp: serverTimestamp(), ...(studentNumber && {studentNumber}) });
                setStep('waiting_approval');
            }
        } catch (error) { 
            console.error("Error submitting student data:", error); 
        } finally {
            setIsSubmittingProfile(false);
        }
    };

    const handleUsePreviousName = async () => {
        try {
            const studentRef = doc(db, 'artifacts', appId, 'public', 'data', 'students', userId);
            await updateDoc(studentRef, { status: 'approved', pendingName: null });
        } catch (e) { console.error(e); }
    };

    const handleTeacherSubmit = () => {
        if (passcode === TEACHER_PASSCODE) onTeacherComplete();
        else setError('Incorrect Passcode.');
    };

    return (
        <div className="fixed inset-0 bg-gray-900 z-[60] flex items-center justify-center p-4">
            <div className="bg-indigo-900/90 border border-indigo-500 p-8 rounded-2xl shadow-2xl max-w-md w-full backdrop-blur-sm text-center relative overflow-hidden">
                {step !== 'role_select' && step !== 'waiting_approval' && step !== 'rejected' && (
                    <button onClick={() => { setStep('role_select'); setError(''); setPasscode(''); }} className="absolute top-4 left-4 text-indigo-300 hover:text-white transition z-10">
                        <ChevronLeft className="w-8 h-8"/>
                    </button>
                )}

                {step === 'role_select' && (
                    <div className="animate-bounce-in">
                        <Globe className="w-16 h-16 mx-auto text-cyan-400 mb-4 animate-pulse" />
                        <h2 className="text-3xl font-black text-white mb-8">Welcome!</h2>
                        
                        <div className="space-y-4">
                            <button onClick={() => setStep('student_setup')} className="w-full bg-cyan-600 hover:bg-cyan-500 text-white font-bold text-xl py-4 rounded-xl shadow-lg transition flex items-center justify-center gap-3">
                                <User className="w-6 h-6"/> Student
                            </button>
                            <button onClick={() => setStep('teacher_login')} className="w-full bg-purple-600 hover:bg-purple-500 text-white font-bold text-xl py-4 rounded-xl shadow-lg transition flex items-center justify-center gap-3">
                                <Key className="w-6 h-6"/> Teacher
                            </button>
                        </div>
                    </div>
                )}

                {step === 'student_setup' && (
                    <div className="animate-bounce-in">
                        <User className="w-16 h-16 mx-auto text-cyan-400 mb-4 animate-pulse" />
                        <h2 className="text-3xl font-black text-white mb-2">Student Setup</h2>
                        <p className="text-indigo-200 mb-6">Enter your real name and group.</p>
                        
                        <input 
                            className="w-full p-4 rounded-xl text-black font-bold text-center text-xl mb-6 focus:ring-4 ring-cyan-400 outline-none"
                            placeholder="Enter Your Full Name"
                            value={name}
                            onChange={e => setName(e.target.value)}
                            disabled={isSubmittingProfile}
                        />
                        
                        <div className="grid grid-cols-2 gap-3 mb-6">
                            {Object.entries(AGE_GROUPS).map(([key, info]) => (
                                <button key={key} onClick={() => setSelectedGroup(key)} disabled={isSubmittingProfile} className={`p-3 rounded-lg border flex flex-col items-center gap-2 transition ${selectedGroup === key ? 'bg-cyan-500 border-cyan-300 text-white shadow-lg scale-105' : 'bg-gray-800 border-gray-600 text-gray-400 hover:bg-gray-700'}`}>
                                    {info.icon}
                                    <span className="text-xs font-bold">{info.label}</span>

                            ))}
                        </div>

const ClassRoster = ({ userId }) => {
    const [students, setStudents] = useState([]);
    const [isOpen, setIsOpen] = useState(true);
    const [nowTime, setNowTime] = useState(Date.now());
    const [autoApprove, setAutoApprove] = useState(false);
   
    useEffect(() => {
        const interval = setInterval(() => setNowTime(Date.now()), 10000); 
        return () => clearInterval(interval);
    }, []);

    useEffect(() => {
        if (!db || !userId) return; 
        const q = query(collection(db, 'artifacts', appId, 'public', 'data', 'students'));
        return onSnapshot(q, (snap) => {
    const now = Date.now();
    const studentList = snap.docs.map(d => {
        const data = d.data();
        const lastPing = data.lastPing;
        if (data.isOnline && lastPing) {
            const pingMs = lastPing.toMillis ? lastPing.toMillis() : (lastPing.seconds * 1000);
            const diffMins = (now - pingMs) / 60000;
            if (diffMins > 2) {
                return { id: d.id, ...data, isOnline: false };
            }
        }
        return { id: d.id, ...data };
    });
    setStudents(studentList);
});
    }, [userId]);

   useEffect(() => {
        if (!db) return;
        const prefRef = doc(db, 'artifacts', appId, 'public', 'data', 'settings', 'preferences');
        const unsub = onSnapshot(prefRef, (snap) => {
            if (snap.exists()) {
                setAutoApprove(snap.data().autoApprove || false);
            }
        });
        return () => unsub();
    }, []);
       const toggleAutoApprove = async (e) => {
        e.stopPropagation();
        const prefRef = doc(db, 'artifacts', appId, 'public', 'data', 'settings', 'preferences');
        await setDoc(prefRef, { autoApprove: !autoApprove }, { merge: true });
    };

    const handleApprove = async (studentId, currentStatus, intendedName) => {
        const studentRef = doc(db, 'artifacts', appId, 'public', 'data', 'students', studentId);
        const studentSnap = await getDoc(studentRef);
        if (!studentSnap.exists()) return;
        const existingData = studentSnap.data();

        let finalNum = existingData.studentNumber;
        if (!finalNum) {
            const approvedStudents = students.filter(s => s.status === 'approved');
            const maxNum = approvedStudents.reduce((max, s) => s.studentNumber > max ? s.studentNumber : max, 0);
            finalNum = maxNum + 1;
        }

        await updateDoc(studentRef, {
            status: 'approved',
            name: intendedName || existingData.name,
            pendingName: null,
            studentNumber: finalNum
        });
    };

    useEffect(() => {
        if (autoApprove && students.length > 0) {
            const pendingStudents = students.filter(s => s.status === 'pending');
            pendingStudents.forEach(s => {
                handleApprove(s.id, s.status, s.pendingName || s.name);
            });

const FloatingStatsBar = ({ rank, totalLessons }) => {
    const [pos, setPos] = useState({ x: null, y: 16 });
    const dragging = useRef(false);
    const offset = useRef({ x: 0, y: 0 });

    const startDrag = (clientX, clientY, el) => {
        dragging.current = true;
        const rect = el.getBoundingClientRect();
        offset.current = { x: clientX - rect.left, y: clientY - rect.top };
    };
    const onMove = (clientX, clientY) => {
        if (!dragging.current) return;
        setPos({ x: clientX - offset.current.x, y: clientY - offset.current.y });
    };
    const stopDrag = () => { dragging.current = false; };

    useEffect(() => {
        const mouseMove = (e) => onMove(e.clientX, e.clientY);
        const touchMove = (e) => { if (e.touches[0]) onMove(e.touches[0].clientX, e.touches[0].clientY); };
        window.addEventListener('mousemove', mouseMove);
        window.addEventListener('touchmove', touchMove);
        window.addEventListener('mouseup', stopDrag);
        window.addEventListener('touchend', stopDrag);
        return () => {
            window.removeEventListener('mousemove', mouseMove);
            window.removeEventListener('touchmove', touchMove);
            window.removeEventListener('mouseup', stopDrag);
            window.removeEventListener('touchend', stopDrag);
        };
    }, []);

    const style = pos.x === null
        ? { left: '50%', top: '16px', transform: 'translateX(-50%)' }
        : { left: pos.x + 'px', top: pos.y + 'px' };

    return (
        <div
            onMouseDown={(e) => startDrag(e.clientX, e.clientY, e.currentTarget)}
            onTouchStart={(e) => e.touches[0] && startDrag(e.touches[0].clientX, e.touches[0].clientY, e.currentTarget)}
            style={{ position: 'fixed', zIndex: 70, cursor: 'grab', ...style }}
            className="bg-gray-800/95 border border-indigo-500 rounded-full shadow-2xl px-5 py-2 flex items-center gap-4 select-none backdrop-blur-md"
        >
            <span className="flex items-center gap-1 text-yellow-400 font-black"><Trophy className="w-4 h-4" /> #{rank || '-'}</span>
            <span className="text-gray-500">|</span>
            <span className="flex items-center gap-1 text-teal-300 font-bold"><BookOpen className="w-4 h-4" /> {totalLessons} Lessons</span>
        </div>
    );
};

        if (bulkDrafts.some(d => !d.title.trim())) { showMessage('Please give every lesson a title first.'); return; }
        setBulkRunning(true); setBulkStage('Saving the lessons...');
        const base = Date.now();
        try {
            for (const [k, d] of bulkDrafts.entries()) {
                const hasImages = imageNamesIn(d.content).length > 0;
                const extra = hasImages ? '' : imageLinesFor(bulkFirstImage, k);
                const content = (d.content.trim() + (extra ? '\n' + extra : '')).trim();
                const ref = doc(collection(db, 'artifacts', appId, 'public', 'data', 'lessons'));
                await setDoc(ref, { title: d.title.trim(), burmeseContent: content, timestamp: new Date(base + k * 1000), variants: {} });
            }
            const s = String(bulkFirstImage || '').trim();
            if (/^\d+$/.test(s) && bulkDrafts.some(d => imageNamesIn(d.content).length === 0)) {
                const next = String(parseInt(s, 10) + 5 * bulkDrafts.filter(d => imageNamesIn(d.content).length === 0).length).padStart(s.length, '0');
                localStorage.setItem('lastImageSeqNum', next); setBulkFirstImage(next);
            }
            bulkAddLog(`SAVED ${bulkDrafts.length} lesson(s).`);
            setBulkDrafts([]);
            showMessage(`Saved ${bulkDrafts.length} lessons. Now press button 3.`);
        } catch (err) { console.error(err); showMessage('Saving failed: ' + err.message); }
        setBulkRunning(false); setBulkStage('');
    };

    // STEP 3: the 4 age versions of every lesson that has not got them, one lesson at a time (one Gemini call per age group), a rest between lessons
    const lessonsWaitingCount = lessons.filter(l => !lessonHasAllVariants(l)).length;
    const bulkGenerateAll = async () => {
        if (bulkRunning) return;
        const todo = lessons.filter(l => !lessonHasAllVariants(l));
        if (!todo.length) { showMessage('Every lesson already has its 4 age versions.'); return; }
        bulkStopRef.current = false; setBulkRunning(true);
        let ok = 0, bad = 0;
        for (const [n, l] of todo.entries()) {
            if (bulkStopRef.current) break;
            const variants = { ...(l.variants || {}) };
            const notes = [];
            try {
                for (const group of ALL_GROUPS) {
                    if (variants[group] && variants[group].burmese && variants[group].english) continue;
                    setBulkStage(`Step 3: lesson ${n + 1} of ${todo.length} "${l.title}" -- ${group}`);
                    const { data, problems } = await generateSingleVariantChecked(l.title, l.burmeseContent, group);
                    variants[group] = data;
                    if (problems.length) notes.push(`${group}: ${problems.join('; ')}`);
                    if (data.quiz) await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'lessons', l.id, 'quiz', group), data.quiz);
                    await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'lessons', l.id), { variants }, { merge: true });   // saved after every age group
                    await sleep(INNER_GAP_MS);
                }
                ok++;
                bulkAddLog(`OK  ${l.title}: ${ALL_GROUPS.map(g => wordCount(plainEnglish(variants[g] && variants[g].english))).join(' / ')} English words${notes.length ? '  -- check: ' + notes.join(' | ') : ''}`);
            } catch (err) {
                bad++;
                bulkAddLog(`FAILED  ${l.title}: ${err.message}  (the age groups that were finished are saved; press the button again to carry on)`);
            }
            if (n < todo.length - 1 && !bulkStopRef.current) { setBulkStage(`Resting ${bulkPauseRef.current} seconds...`); await sleep(bulkPauseRef.current * 1000); }
        }
        setBulkRunning(false); setBulkStage('');
        showMessage(bulkStopRef.current ? `Stopped. ${ok} lesson(s) done.` : `Finished: ${ok} lesson(s) done${bad ? `, ${bad} failed (press the button again)` : ''}.`);
    };

    const bulkApi = {
        drafts: bulkDrafts, running: bulkRunning, stage: bulkStage, log: bulkLog,
        pauseSec: bulkPause, setPauseSec: setBulkPause, firstImageNum: bulkFirstImage, setFirstImageNum: setBulkFirstImage,
        onFiles: bulkOnFiles, setTitle: bulkSetTitle, removeDraft: bulkRemove, clearDrafts: () => setBulkDrafts([]),
        saveAll: bulkSaveAll, generateAll: bulkGenerateAll, stop: bulkStop, lessonsWaiting: lessonsWaitingCount,
    };
    // ======================= end of BULK LESSONS =======================

    const handleGenerateVariants = async (lesson, mode) => {
        setIsLoading(true); 
        setGeneratingLessonId(lesson.id);
        
        try {
            const newVariants = await generateLessonVariants(lesson.title, lesson.burmeseContent, mode);
            await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'lessons', lesson.id), { variants: newVariants }, { merge: true });
            for (const [group, data] of Object.entries(newVariants)) {
                if (data.quiz) await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'lessons', lesson.id, 'quiz', group), data.quiz);
            }
            showMessage("Generated Successfully!");
        } catch (e) { 
            showMessage("Error generating."); 
            console.error(e); 
        } finally { 
            setIsLoading(false); 
            setGeneratingLessonId(null);
        }
    };

    const handleExportLessons = async () => {
        setIsLoading(true); showMessage("Exporting Full Database...");
        try {
            const lessonsSnap = await getDocs(collection(db, 'artifacts', appId, 'public', 'data', 'lessons'));
            const scoresSnap = await getDocs(collection(db, 'artifacts', appId, 'public', 'data', 'global_scores'));
            const studentsSnap = await getDocs(collection(db, 'artifacts', appId, 'public', 'data', 'students'));
            const activityFeedSnap = await getDocs(collection(db, 'artifacts', appId, 'public', 'data', 'activity_feed'));

            const toMillisSafely = (ts) => {
                if (!ts) return null;
                if (typeof ts.toMillis === 'function') return ts.toMillis();
                if (ts.seconds) return ts.seconds * 1000;
                return null;
            };

            const exportData = {
                version: 3,
                lessons: [],
                globalScores: scoresSnap.docs.map(d => ({ id: d.id, ...d.data(), timestamp: toMillisSafely(d.data().timestamp) })),
                students: studentsSnap.docs.map(d => ({ id: d.id, ...d.data(), timestamp: toMillisSafely(d.data().timestamp), lastSeen: toMillisSafely(d.data().lastSeen) })),
                activityFeed: activityFeedSnap.docs.map(d => ({ id: d.id, ...d.data(), timestamp: toMillisSafely(d.data().timestamp) })),
                questions: {},
                quizResults: {}
            };

            for (const d of lessonsSnap.docs) {
                const lData = d.data();
                exportData.lessons.push({ id: d.id, ...lData, timestamp: toMillisSafely(lData.timestamp) });

                const qSnap = await getDocs(collection(db, 'artifacts', appId, 'public', 'data', 'lessons', d.id, 'questions'));
                if (!qSnap.empty) {
                    exportData.questions[d.id] = qSnap.docs.map(q => ({ id: q.id, ...q.data(), timestamp: toMillisSafely(q.data().timestamp) }));
                }

                const groups = ['storytellers', 'explorers', 'adventurers', 'voyagers'];
                for (const g of groups) {
                    const rSnap = await getDocs(collection(db, 'artifacts', appId, 'public', 'data', 'lessons', d.id, 'quiz', g, 'results'));
                    if (!rSnap.empty) {
                        exportData.quizResults[`${d.id}_${g}`] = rSnap.docs.map(r => ({ id: r.id, ...r.data(), timestamp: toMillisSafely(r.data().timestamp) }));
                    }
                }
            }

            const blob = new Blob([JSON.stringify(exportData, null, 2)], { type: "application/json" });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url; a.download = `full_app_backup_${new Date().toISOString().slice(0, 10)}.json`;
            document.body.appendChild(a); a.click(); document.body.removeChild(a);
            showMessage("Exported Successfully!");
        } catch(e) { console.error(e); showMessage("Error Exporting!"); } finally { setIsLoading(false); }
    };

    const handleImportLessons = (event) => {
        const files = Array.from(event.target.files || []);
        const file = files[0];
        if (!file) return;
        // A lessons file (several files, .md / .txt, or the .json that the make-lessons tool writes) goes into the "Bulk lessons" box below,
        // where the titles are checked before saving. The old full-backup .json (made by Export Data) is restored as before.
        const toBulk = (list) => { const n = bulkAddFromTexts(list, list.length); if (n) showMessage(`${n} lesson(s) loaded into the "Bulk lessons" box below. Check the titles, then press 2) Save all.`); };
        if (files.length > 1 || !/\.json$/i.test(file.name)) {
            files.sort((a, b) => a.name.localeCompare(b.name, undefined, { numeric: true, sensitivity: 'base' }));
            Promise.all(files.map(f => f.text().then(t => ({ name: f.name, text: t })))).then(toBulk);
            event.target.value = '';
            return;
        }
        const reader = new FileReader();
        reader.onload = async (e) => {
            try {
                const importedData = JSON.parse(e.target.result);
                if (!Array.isArray(importedData) && importedData.version === undefined && Array.isArray(importedData.lessons)) {
                    toBulk([{ name: file.name, text: e.target.result }]);
                    event.target.value = '';
                    return;
                }
                setIsLoading(true); 
                let lCount = 0, sCount = 0, stuCount = 0, qCount = 0, qrCount = 0;
                
                const toDate = (ts) => ts ? new Date(ts) : serverTimestamp();

                if (importedData.version >= 3) {
                    for (const l of (importedData.lessons || [])) {
                        const { id, timestamp, ...rest } = l;
                        await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'lessons', id), { ...rest, timestamp: toDate(timestamp) });
                        lCount++;
                    }
                    for (const s of (importedData.globalScores || [])) {
                        const { id, timestamp, ...rest } = s;
                        await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'global_scores', id), { ...rest, timestamp: toDate(timestamp) });
                        sCount++;
                    }
                    for (const stu of (importedData.students || [])) {
                        const { id, timestamp, lastSeen, ...rest } = stu;
                        await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'students', id), { ...rest, timestamp: toDate(timestamp), lastSeen: toDate(lastSeen), isOnline: false, currentLesson: null });
                        stuCount++;
                    }
                    for (const act of (importedData.activityFeed || [])) {
                        const { id, timestamp, ...rest } = act;
                        await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'activity_feed', id), { ...rest, timestamp: toDate(timestamp) });
                    }
                    for (const [lessonId, qList] of Object.entries(importedData.questions || {})) {
                        for (const q of qList) {
                            const { id, timestamp, ...rest } = q;
                            await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'lessons', lessonId, 'questions', id), { ...rest, timestamp: toDate(timestamp) });
                            qCount++;
                        }
                    }
                    for (const [key, rList] of Object.entries(importedData.quizResults || {})) {
                        const [lessonId, group] = key.split('_');
                        for (const r of rList) {
                            const { id, timestamp, ...rest } = r;
                            await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'lessons', lessonId, 'quiz', group, 'results', id), { ...rest, timestamp: toDate(timestamp) });
                            qrCount++;
                        }
                    }
                    showMessage(`Imported: ${lCount} lessons, ${stuCount} students, ${qCount} Qs.`);
                } else if (importedData.version === 2) {
                    for (const l of (importedData.lessons || [])) {
                        const { id, timestamp, ...rest } = l;
                        await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'lessons', id), { ...rest, timestamp: toDate(timestamp) });
                        lCount++;
                    }
                    for (const s of (importedData.globalScores || [])) {
                        const { id, timestamp, ...rest } = s;
                        await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'global_scores', id), { ...rest, timestamp: toDate(timestamp) });
                        sCount++;
                    }
                    showMessage(`Imported ${lCount} lessons & ${sCount} scores.`);
                } else {
                    let baseTime = Date.now();
                    for (const item of (Array.isArray(importedData) ? importedData : [])) {
                        if (item.title && item.burmeseContent) {
                            await addDoc(collection(db, 'artifacts', appId, 'public', 'data', 'lessons'), { title: item.title, burmeseContent: item.burmeseContent, timestamp: new Date(baseTime), variants: {} });
                            baseTime += 1000;
                            lCount++;
                        }
                    }
                    showMessage(`Imported ${lCount} lessons.`);
                }
            } catch (error) { showMessage(`Error: ${error.message}`); } finally { setIsLoading(false); }
            event.target.value = ''; 
        };
        reader.readAsText(file);
    };

    const handleStudentOnboarding = (profile) => {
        setStudentProfile(profile);
        localStorage.setItem(`student_profile_${userId}`, JSON.stringify(profile));
    };

    if (!isAuthReady) return <div className="min-h-screen bg-gray-900 flex justify-center items-center text-teal-400"><RotateCw className="animate-spin w-8 h-8"/></div>;

    return (
        <div className="min-h-screen bg-gray-900 text-white font-sans p-4 sm:p-8">
            <script src="https://cdn.tailwindcss.com"></script>
            <style>{`@import url('https://fonts.googleapis.com/css2?family=Inter:wght@400;600;800&display=swap'); * { font-family: 'Inter', sans-serif; } .animate-bounce-in { animation: bounceIn 0.5s ease-out; } @keyframes bounceIn { 0% { transform: scale(0.5); opacity: 0; } 80% { transform: scale(1.05); opacity: 1; } 100% { transform: scale(1); } } .custom-scrollbar::-webkit-scrollbar { width: 8px; } .custom-scrollbar::-webkit-scrollbar-thumb { background: #4B5563; border-radius: 4px; }`}</style>
            
            {!isTeacher && (!studentProfile || studentProfile.status !== 'approved') && (
                <WelcomeModal 
                    userId={userId}
                    onStudentComplete={handleStudentOnboarding} 
                    onTeacherComplete={() => { setIsTeacher(true); setCurrentRole('Teacher'); localStorage.setItem('isTeacher', 'true'); showMessage("Successfully logged in as Teacher."); }} 
                />
            )}

            {/* Student Network Offline Warning */}
            {showStudentWarning && (
                <div className="fixed top-20 left-1/2 transform -translate-x-1/2 bg-red-900/95 text-white px-6 py-4 rounded-xl shadow-2xl z-[100] font-bold animate-pulse flex items-center gap-3 border border-red-500 backdrop-blur-md w-[90%] max-w-md">
                    <div className="w-4 h-4 bg-red-500 rounded-full animate-ping flex-shrink-0"></div>
                    <span className="text-sm">Warning: You have been offline for 3 minutes! Please check your internet connection to reconnect.</span>
                </div>
            )}

            {activeQuizLessonId && activeQuizData && <QuizModule lessonTitle={currentLessonViewing} lessonId={activeQuizLessonId} userId={userId} userName={studentProfile?.name || 'Student'} ageGroup={studentProfile?.group} quizData={activeQuizData} onClose={() => { setActiveQuizLessonId(null); setActiveQuizData(null); }} />}
            {showGlobalLeaderboard && <LeaderboardView leaderboard={globalLeaderboardData} title="Global Champion Board" loading={loadingGlobalLB} onClose={() => setShowGlobalLeaderboard(false)} isGlobal={true} currentUserId={userId} />}

            <CompletionToastStack toasts={completionToasts} />
{currentRole === 'Student' && studentProfile?.status === 'approved' && (
    <FloatingStatsBar rank={studentRank} totalLessons={studentTotalLessons} />
)}

            <div className="max-w-4xl mx-auto">
                <header className="mb-6 flex flex-col md:flex-row gap-4 justify-between items-center bg-gray-800 p-4 rounded-xl shadow-lg border border-gray-700 z-50 relative">
                   <div className="flex flex-wrap items-center gap-2">
                        {isTeacher ? (
                            <div className="flex gap-2">
                                <button onClick={() => setCurrentRole(currentRole === 'Teacher' ? 'Student' : 'Teacher')} className={`px-4 py-2 font-bold rounded shadow-lg transform transition hover:scale-105 ${currentRole === 'Teacher' ? 'bg-purple-600 hover:bg-purple-700' : 'bg-teal-600 hover:bg-teal-700'}`}>
                                    {currentRole === 'Teacher' ? 'Switch to Student View' : 'Switch to Teacher View'}
                                </button>
                                <button onClick={handleTeacherLogout} className="px-4 py-2 font-bold rounded shadow-lg bg-red-600 hover:bg-red-700 transition">Teacher Logout</button>
                            </div>
                        ) : (
                            <button onClick={handleTeacherLogin} className="px-4 py-2 font-bold rounded shadow-lg bg-gray-600 hover:bg-gray-500 transition flex items-center gap-2"><Key className="w-4 h-4"/> Teacher Login</button>
                        )}

                        {studentProfile && studentProfile.status === 'approved' && (
                            <div className="flex items-center gap-3 ml-2">
                                <span className="bg-gray-700 px-3 py-1 rounded text-gray-300 font-medium flex items-center gap-2"><User className="w-4 h-4"/> {studentProfile.name} <span className="text-xs bg-indigo-500 text-white rounded px-1">#{studentProfile.studentNumber}</span></span>
                                <span className="bg-indigo-900/50 border border-indigo-500/30 px-3 py-1 rounded text-indigo-300 text-xs uppercase font-bold flex items-center gap-1">{AGE_GROUPS[studentProfile.group]?.icon} {AGE_GROUPS[studentProfile.group]?.label}</span>
                                <button onClick={() => setShowGlobalLeaderboard(true)} className="p-2 bg-yellow-600/20 text-yellow-400 border border-yellow-500/50 rounded-full hover:bg-yellow-600/40 transition hover:scale-110 active:scale-90 shadow-[0_0_10px_rgba(234,179,8,0.3)]" title="View Global Leaderboard"><Trophy className="w-5 h-5" /></button>
                            </div>
                        )}
                   </div>
                   <div className="flex items-center gap-4">
                       <div className="text-yellow-400 text-xs font-bold bg-gray-900 px-2 py-1 rounded border border-yellow-600">Active folder: {IMG_BASE_URL.split('/')[4]}</div>
                       <div className="text-gray-400 text-sm font-semibold tracking-wide">{currentRole} View</div>
                       
                       {/* Notification Bell Integrated into Header Right Side */}
                       <NotificationBell userId={userId} />
                   </div>
                </header>

                {message && <div className="fixed top-4 left-1/2 transform -translate-x-1/2 bg-blue-600 text-white px-6 py-2 rounded-full shadow-xl z-50 font-bold animate-bounce">{message}</div>}

                {currentRole === 'Teacher' ? (
                    <div className="space-y-6">
                        <ClassRoster userId={userId} />

                        <div className="bg-gray-800 p-6 rounded-xl shadow-xl border border-gray-700">
                            <div className="flex justify-between items-center mb-4">
                                <h3 className="text-xl font-bold text-teal-300">{editingLesson ? 'Edit Lesson' : 'Add New Lesson'}</h3>
                                <div className="flex gap-2">
                                     <label htmlFor="import-lessons" className="cursor-pointer bg-indigo-600 p-2 rounded hover:bg-indigo-700 transition flex items-center gap-1 text-sm" title="Import a lessons file (.json from make-lessons, or .md / .txt articles) or a full backup (.json from Export Data)">
                                        <input type="file" id="import-lessons" accept=".json,.md,.txt" multiple onChange={handleImportLessons} className="hidden" />
                                        <Upload className="w-4 h-4 text-white" /> Import Data / Lessons file
                                     </label>
                                     <button onClick={handleExportLessons} className="bg-pink-600 p-2 rounded hover:bg-pink-700 transition flex items-center gap-1 text-sm" title="Export JSON"><Download className="w-4 h-4 text-white" /> Export Data</button>
                                </div>
                            </div>
                            <form onSubmit={handleSaveLesson} className="space-y-4">
                                <input value={newLessonTitle} onChange={e => setNewLessonTitle(e.target.value)} placeholder="Lesson Title" className="w-full p-3 bg-gray-900 border border-gray-600 rounded text-white focus:border-teal-500 focus:outline-none" disabled={isLoading}/>
                                <textarea value={newLessonContent} onChange={e => setNewLessonContent(e.target.value)} placeholder="Content (e.g., insert 000101.png)" rows="6" className="w-full p-3 bg-gray-900 border border-gray-600 rounded text-white focus:border-teal-500 focus:outline-none" disabled={isLoading}/>
                                
                                <div className="flex gap-2 items-center bg-gray-900 p-2 rounded border border-gray-600">
                                    <ImageIcon className="w-4 h-4 text-gray-400 ml-2" />
                                    <input type="text" placeholder="e.g. 001 or 000101" value={imageInput} onChange={e => setImageInput(e.target.value)} className="bg-transparent text-white text-sm focus:outline-none w-32 border-b border-gray-600 focus:border-teal-500" />
                                    <span className="text-gray-500 text-xs">auto-generates 5 .png images</span>
                                    <button type="button" onClick={insertImageSequence} className="text-xs bg-gray-700 hover:bg-gray-600 px-3 py-1 rounded text-white">Add</button>
                                </div>

                                <div className="flex gap-2">
                                    <button type="submit" className="flex-1 bg-teal-600 p-3 rounded hover:bg-teal-700 flex justify-center items-center font-bold shadow-lg transition" disabled={isLoading}>
                                        {isLoading && !generatingLessonId ? <RotateCw className="animate-spin w-5 h-5 mr-2"/> : (editingLesson ? <Edit2 className="w-5 h-5 mr-2"/> : <BookOpen className="w-5 h-5 mr-2"/>)}{editingLesson ? 'Update' : 'Create & Save'}
                                    </button>
                                    {editingLesson && <button type="button" onClick={() => { setEditingLesson(null); setNewLessonTitle(''); setNewLessonContent(''); }} className="bg-gray-600 p-3 rounded hover:bg-gray-500 transition">Cancel</button>}
                                </div>
                            </form>
                        </div>
                        <BulkLessonsCard bulk={bulkApi} />
                        <div className="space-y-4">
                            {lessons.map(l => (
                                <LessonItem key={l.id} lesson={l} isTeacher={true} onGenerateVariants={handleGenerateVariants} onEdit={handleEditLesson} isGenerating={generatingLessonId === l.id} userId={userId} isOpen={openLessonId === l.id} onToggle={() => setOpenLessonId(openLessonId === l.id ? null : l.id)} />
                            ))}
                        </div>
                    </div>
                ) : (
                    <div className="space-y-6">
                        {lessons.map(l => (
                            <LessonItem key={l.id} lesson={l} isTeacher={false} studentAgeGroup={studentProfile?.group} studentName={studentProfile?.name} onTakeQuiz={(id, title, data) => { setActiveQuizLessonId(id); setActiveQuizData(data); }} isGenerating={false} userId={userId} isOpen={openLessonId === l.id} onToggle={() => setOpenLessonId(openLessonId === l.id ? null : l.id)} />
                        ))}
                        {lessons.length === 0 && <div className="text-center text-gray-500 py-10">No lessons available yet.</div>}
                    </div>
                )}
            </div>
        </div>
    );
};

export default App;
